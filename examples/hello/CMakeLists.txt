cmake_minimum_required(VERSION 3.24)
project(ModlockHello LANGUAGES CXX)

find_package(Modlock CONFIG REQUIRED)
option(SAMPLE_ABI_MISMATCH "Build an intentionally incompatible test plugin" OFF)
option(SAMPLE_FAIL_START "Exercise cleanup after partial plugin startup" OFF)
add_library(hello MODULE hello.cc)
target_link_libraries(hello PRIVATE modlock::sdk)
target_compile_features(hello PRIVATE cxx_std_23)
if(SAMPLE_ABI_MISMATCH)
  target_compile_definitions(hello PRIVATE SAMPLE_ABI_MISMATCH)
endif()
if(SAMPLE_FAIL_START)
  target_compile_definitions(hello PRIVATE SAMPLE_FAIL_START)
endif()
